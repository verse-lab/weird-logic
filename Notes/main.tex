\left\{ 
% \begin{array}{l} V_C \\ V_L \end{array}

For any program \texttt{l} written in the language \texttt{L}, there always exists a payload \texttt{p} such that the program \texttt{C} and \texttt{l} execute the same behaviour.

\subsection{\texttt{if-else}}
Given a Doppler attack language $L$:

\[
\begin{array}{rrl}
     L & \rightarrow & ~~S_1 : x=x+1 \\
       &   & |~S_2 : x=x+2 \\
\end{array}
\]

and vulnerable program $C$:
\begin{lstlisting}
choose a
if a then
    x = x+1
else
    x = x+2
\end{lstlisting}

write the triple: 
\[
\left\{ x(\mathcal{P}) = x(L) \right\}
\left[
\begin{array}{ll}
p \in \mathcal{P} & , ~C(p) \\
l \in L & , ~l
\end{array}
\right]
\left\{
\begin{array}{c}
\forall l \in L , \exists p \in \mathcal{P} \\
x(p) = x(l)
\end{array}
\right\}
\]

where $P = \{a_\top, a_{\bot} \}$,

\[
% \left\{ \forall l, p, x(p) = x(l) \right\}
\left\{ x(\mathcal{P})=x(L) \right\}
\left[
\begin{array}{ll}
p \in \{a_\top, a_{\bot}\} & , ~C(p) \\
l \in S_1 \mid S_2 & , ~l
\end{array}
\right]
\left\{
\begin{array}{c}
\forall l \in L, \exists p \in \mathcal{P} \\
x(p) = x(l)
\end{array}
\right\}
\]

Apply GrmDisj-rule:


\[
\quad
\frac{
\begin{array}{c}
L \rightarrow L_1 \mid L_2, L \notin L_1, L \notin L_2 \\
\{ P \} \ [C(\mathcal{P}) ; L_1] \ \{ Q\} ~~~~
\{ P \} \ [C(\mathcal{P}) ; L_2] \ \{ Q\} \\
\end{array}
}{
\{ P \} \ [ C(\mathcal{P}); L] \ \{ Q \}
}
\]

% conflicts of symbols (L)

Partition $S_1$ and $S_2$:

\[
\forall i \in \{1, 2\}, 
\left\{ x(\mathcal{P})=x(L) \right\}
% \left\{  \forall l, p, x(p) = x(l) \right\}
\left[
\begin{array}{ll}
p \in \{a_\top, a_\bot\} & , ~C(p) \\
l \in S_i & , ~l
\end{array}
\right]
\left\{
\begin{array}{c}
\forall l \in S_i, \exists p \in \mathcal{P} \\
x(p) = x(l)
\end{array}
\right\}
\]

Apply Weaken-rule:
\[
\quad
\frac{
\begin{array}{c}
 \mathcal{P} \supseteq \mathcal{P}_1 ~~~~
\{ P \} \ [C(\mathcal{P}_1) ; L] \ \{ Q\} \\
\end{array}
}{
\{ P \} \ [ C(\mathcal{P}); L] \ \{ Q \}
}
\]

Let $\mathcal{P}_1 = \{a_\top\}; \mathcal{P}_2 = \{a_\bot\}$, shrink $\mathcal{P}$:

\[
\forall i \in \{1, 2\}, 
\left\{ x(\mathcal{P})=x(L) \right\}
% \left\{  \forall l, p, x(p) = x(l) \right\}
\left[
\begin{array}{ll}
p \in  \mathcal{P}_i& , ~C(p) \\
l \in S_i & , ~l
\end{array}
\right]
\left\{
\begin{array}{c}
\forall l \in S_i, \exists p \in \mathcal{P}_i \\
x(p) = x(l)
\end{array}
\right\}
\]

Apply lang-rule:

\[
\quad
\frac{
\begin{array}{c}
 \exists l , L = \{l\}~~~~
~\{ P \} \ [C(\mathcal{P}) ; l] \ \{ \exists p \in \mathcal{P}, Q(l,p)\} \\
\end{array}
}{
\{ P \} \ [ C(\mathcal{P}); L] \ \{ \forall l \in L, \exists p \in \mathcal{P}, Q(l,p) \}
}
\]

Apply payload-rule:

\[
\quad
\frac{
\begin{array}{c}
\exists p \in \mathcal{P},~\{ P \} \ [C(p) ; l] \ \{ Q(l,p)\} \\
\end{array}
}{
\{ P \} \ [ C(\mathcal{P}); l] \ \{ \exists p \in \mathcal{P}, Q(l,p) \}
}
\]

\subsection{\texttt{while-choose}}
Given a Doppler attack language $L$:

\[
\begin{array}{rrl}
     L & \rightarrow & ~~S_1 \\
     S_1 & \rightarrow & ~~x=x+1;S_1 \\
     &  & \mid \epsilon
\end{array}
\]

or vulnerable program $C$:
\begin{lstlisting}
while (choose b) do
    x = x+1
\end{lstlisting}

write the triple for $C$: 
\[
\left\{ \forall l, p, x(p) = x(l) \right\}
% \left\{ x(\mathcal{P})=x(L) \right\}
\left[
\begin{array}{ll}
p \in \mathcal{P} & , ~C(p) \\
l \in L & , ~l
\end{array}
\right]
\left\{
\begin{array}{c}
\forall l \in L , \exists p \in \mathcal{P} \\
x(p) = x(l)
\end{array}
\right\}
\]

where $\mathcal{P} := \mathbb{B}^*$

Apply weaken-rule and shrink $\mathcal{P}$ to $\{b_\top*;b_\bot\}$

Apply LeftReg-rule:
\[
\quad
\frac{
\begin{array}{c}
 \mathcal{P} = \{p_\top*;p_\bot\}\\
\{ P \wedge \mathcal{P}_1=\{p_\top\}\} \ [C'(\mathcal{P}_1) ; L_1] \ \{ P\} \\
\end{array}
}{
\{ P \} \ \left[ 
\begin{array}{c}
     \texttt{while} ~\mathcal{P}~\texttt{do}~C'  \\
     L\rightarrow (L_1;L)\mid \epsilon
\end{array}\right] \ 
\{ P \wedge \exists p', p =p';p_\bot \}
}
{\text{LeftReg}}
\]

Seq-rule (should have more constraints):
\[
\quad 
\frac{
\{P\}\left[C_1(\mathcal{P}_1),~L_1\right]\{H\} ~~
\{H\}\left[C_2(\mathcal{P}_2),~L_2\right]\{Q\}
}
{
\{P\}\left[
\begin{array}{ll}
     p \in \mathcal{P}_1 \times \mathcal{P}_2 &, C_1(p_1);C_2(p_2) \\
     l \in L_1;L_2&, l 
\end{array}\right]\{Q\}
}
{Seq}
\]

SeqLeft-rule:
\[
\quad
\frac{
\begin{array}{c}
\{ P \} \ [C_1(\mathcal{P}_1) ; L\rightarrow (L_1;L)\mid \epsilon] \ \{ H\} \\
\{ H \} \ [C_2(\mathcal{P}_2) ; L\rightarrow L_2] \ \{ Q\} \\
\end{array}
}{
\{ P \} \ \left[ 
\begin{array}{ll}
     p \in \mathcal{P}_1 \times \mathcal{P}_2 &, C_1(p_1);C_2(p_2) \\

\end{array}\right] \

}
{\text{SeqLeft}}
\]

SeqRight-rule:
\[
\quad
\frac{
\begin{array}{c}

\end{array}

\{ P \} \ \left[

     p \in \mathcal{P}_1 \times \mathcal{P}_2 &, C_1(p_1);C_2(p_2) \\

\subsection{\texttt{do-while}}
Given a Doppler attack language $L$:

\[
\begin{array}{rrl}
     L & \rightarrow & ~~S_1 \\
     S_1 & \rightarrow & ~~x=x+1 \\
     &  & \mid S_1;S_1
\end{array}
\]
or language $L'$:
\[
\begin{array}{rrl}
     L & \rightarrow & ~~x=x+1;S_1 \\
     S_1 & \rightarrow & ~~x=x+1; S_1 \\
     &  & \mid \epsilon
\end{array}
\]

and vulnerable program $C$:
\begin{lstlisting}
do
    choose a
    x = x+1
while a
\end{lstlisting}

or vulnerable program $C'$:
\begin{lstlisting}
b=1
while b>0 do
    b--
    choose b
    x = x+1
\end{lstlisting}

or vulnerable program $C''$:
\begin{lstlisting}
c=true
while c do
    choose c
    x = x+1
\end{lstlisting}
TODO: The mapping between program and language is not isomorphic. One solution is to restrict the language to left-regular format when it can be converted.
For example, restrict the language to $L'$ as shown above.

Write the triple for $C$:
\[
% \left\{ \forall l, p, x(p) = x(l) \right\}
\left\{ x(\mathcal{P})=x(L) \right\}
\left[
\begin{array}{ll}
p \in \{\epsilon\} \times \mathcal{P}' & , ~C_1(\epsilon);C_2(p') \\
l \in \{L \rightarrow x=x+1;S_1\} & , ~l
\end{array}
\right]
\left\{
\begin{array}{c}
\forall l \in L , \exists p \in \mathcal{P} \\
x(p) = x(l)
\end{array}
\right\}
\]
where $\mathcal{P}:=\mathbb{B}*$, $\mathcal{P} := \{\epsilon\} \times \mathcal{P}'$, and $p' \in \mathcal{P}'$; 

$C_1 := x=x+1$; 

$C_2 := \texttt{while}~choose~a~\texttt{do}~x=x+1$

Apply Seq-rule:
\[
(1)
% \left\{ \forall l, p, x(p) = x(l) \right\}
\left\{ x(\mathcal{P})=x(L_1) \right\}
\left[
\begin{array}{ll}
p \in \{\epsilon\} & , ~C_1(p) \\
l \in \{L_1 \rightarrow x=x+1\} & , ~l
\end{array}
\right]
\left\{
\begin{array}{c}
\forall l \in L_1 , \exists p \in \mathcal{P} \\
x(p) = x(l)
\end{array}
\right\}
\]

\[
(2)
% \left\{ \forall l, p, x(p) = x(l) \right\}
\left\{ x(\mathcal{P}')=x(S_1) \right\}
\left[
\begin{array}{ll}
p \in \mathcal{P}' & , C_2(\mathcal{P}') \\
l \in \{S_1 \rightarrow x=x+1;S_1|\epsilon\} & , ~l
\end{array}
\right]
\left\{
\begin{array}{c}
\forall l \in S_1 , \exists p \in \mathcal{P}' \\
x(p) = x(l)
\end{array}
\right\}
\]

(2) is the same as the \texttt{while-choose} example.

\subsection{\texttt{choose-while}}

\[
\begin{array}{rrl}

     S_1 & \rightarrow & ~~x=x+1 ; S_1\\

\end{array}
\]

\[
\begin{array}{rrl}
     L & \rightarrow & \texttt{for}~i~\texttt{in}~b~\texttt{do}~S_1\\

\end{array}
\]

and vulnerable program $C$:
\begin{lstlisting}

    b--

\subsection{context-free}
Given a Doppler attack language $L$:

\[
\begin{array}{rrl}
     L & \rightarrow & ~~S_1 \\
     S_1 & \rightarrow & ~~x=x+1; S_1; y=y+1 \\
     & & \mid \epsilon
\end{array}
\]

or in the form of $L=\{(x=x+1)^n;(y=y+1)^n \mid n \geq 0\}$

and vulnerable program $C$:
\begin{lstlisting}
choose n
a=n; b=n
while a>0
    a--
    x = x+1
while b>0
    b--
    y = y+1
\end{lstlisting}

Write the triple for $C$:
\[
% \left\{ \forall l, p, x(p) = x(l) \right\}
\Gamma \vdash \left\{ x(\mathcal{P})=x(L) \wedge y(\mathcal{P}) = y(L)\right\}
\left[
\begin{array}{ll}
p \in \mathcal{P} & , ~C(p) \\
l \in L & , ~l
\end{array}
\right]
\left\{
\begin{array}{c}
\forall l \in L , \exists p \in \mathcal{P} \\
x(p) = x(l) \wedge y(p) = y(l)
\end{array}
\right\}
\]
where $\mathcal{P}:=\mathbb{N}$ and $\Gamma := \{a=b\}$.
