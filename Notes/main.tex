\section{Examples}
Correctness of Doppler in hyper triple:
\[
\{ \forall x \in Var, x(\mathcal{P}) = x(L) \} 
\left[ \begin{array}{l} p \in \mathcal{P} : C(p) \\ l \in L : l\end{array} \right] 
\left\{ 
% \begin{array}{l} V_C \\ V_L \end{array}
% \ \middle|\
\begin{array}{l}
~ \forall ~l \in L, \exists ~p \in \mathcal{P} \\
~ x(p) = x(l) \end{array}\right\}
\]

For any program \texttt{l} written in the language \texttt{L}, there always exists a payload \texttt{p} such that the program \texttt{C} and \texttt{l} execute the same behaviour.

\subsection{\texttt{if-else}}
Given a Doppler attack language $L$:

\[
\begin{array}{rrl}
     L & \rightarrow & ~~S_1 : x=x+1 \\
       &   & |~S_2 : x=x+2 \\
\end{array}
\]

and vulnerable program $C$:
\begin{lstlisting}
choose a
if a then
    x = x+1
else
    x = x+2
\end{lstlisting}

write the triple: 
\[
\left\{ x(\mathcal{P}) = x(L) \right\}
\left[
\begin{array}{ll}
p \in \mathcal{P} & , ~C(p) \\
l \in L & , ~l
\end{array}
\right]
\left\{
\begin{array}{c}
\forall l \in L , \exists p \in \mathcal{P} \\
x(p) = x(l)
\end{array}
\right\}
\]

where $P = \{a_\top, a_{\bot} \}$,

\[
% \left\{ \forall l, p, x(p) = x(l) \right\}
\left\{ x(\mathcal{P})=x(L) \right\}
\left[
\begin{array}{ll}
p \in \{a_\top, a_{\bot}\} & , ~C(p) \\
l \in S_1 \mid S_2 & , ~l
\end{array}
\right]
\left\{
\begin{array}{c}
\forall l \in L, \exists p \in \mathcal{P} \\
x(p) = x(l)
\end{array}
\right\}
\]

Apply GrmDisj-rule:


\[
\quad
\frac{
\begin{array}{c}
L = L_1 \mid L_2 \\
\{ P \} \ [C(\mathcal{P}) ; L_1] \ \{ Q\} ~~~~
\{ P \} \ [C(\mathcal{P}) ; L_2] \ \{ Q\} \\
\end{array}
}{
\{ P \} \ [ C(\mathcal{P}); L] \ \{ Q \}
}
\]

Partition $S_1$ and $S_2$:

\[
\forall i \in \{1, 2\}, 
\left\{ x(\mathcal{P})=x(L) \right\}
% \left\{  \forall l, p, x(p) = x(l) \right\}
\left[
\begin{array}{ll}
p \in \{a_\top, a_\bot\} & , ~C(p) \\
l \in S_i & , ~l
\end{array}
\right]
\left\{
\begin{array}{c}
\forall l \in S_i, \exists p \in \mathcal{P} \\
x(p) = x(l)
\end{array}
\right\}
\]

Apply Weaken-rule:
\[
\quad
\frac{
\begin{array}{c}
 \mathcal{P} \supseteq \mathcal{P}_1 ~~~~
\{ P \} \ [C(\mathcal{P}_1) ; L] \ \{ Q\} \\
\end{array}
}{
\{ P \} \ [ C(\mathcal{P}); L] \ \{ Q \}
}
\]

Let $\mathcal{P}_1 = \{a_\top\}; \mathcal{P}_2 = \{a_\bot\}$, shrink $\mathcal{P}$:

\[
\forall i \in \{1, 2\}, 
\left\{ x(\mathcal{P})=x(L) \right\}
% \left\{  \forall l, p, x(p) = x(l) \right\}
\left[
\begin{array}{ll}
p \in  \mathcal{P}_i& , ~C(p) \\
l \in S_i & , ~l
\end{array}
\right]
\left\{
\begin{array}{c}
\forall l \in S_i, \exists p \in \mathcal{P}_i \\
x(p) = x(l)
\end{array}
\right\}
\]

Apply lang-rule:

\[
\quad
\frac{
\begin{array}{c}
 \exists l , L = \{l\}~~~~
~\{ P \} \ [C(\mathcal{P}) ; l] \ \{ \exists p \in \mathcal{P}, Q(l,p)\} \\
\end{array}
}{
\{ P \} \ [ C(\mathcal{P}); L] \ \{ \forall l \in L, \exists p \in \mathcal{P}, Q(l,p) \}
}
\]

Apply payload-rule:

\[
\quad
\frac{
\begin{array}{c}
\exists p \in \mathcal{P},~\{ P \} \ [C(p) ; l] \ \{ Q(l,p)\} \\
\end{array}
}{
\{ P \} \ [ C(\mathcal{P}); l] \ \{ \exists p \in \mathcal{P}, Q(l,p) \}
}
\]

\subsection{\texttt{choose-while}}

\[

     S_1 & \rightarrow & ~~x=x+1 ; S_1\\

\end{array}
\]

and vulnerable program $C$:
\begin{lstlisting}

    b--
    x = x+1
\end{lstlisting}

or vulnerable program $C'$:
\begin{lstlisting}

    b = b-2;

\end{lstlisting}

\[
% \left\{ \forall l, p, x(p) = x(l) \right\}

\end{array}

\subsection{context-free}
Given a Doppler attack language $L$:

\[
\begin{array}{rrl}
     L & \rightarrow & ~~S_1 \\
     S_1 & \rightarrow & ~~x=x+1; S_1; y=y+1 \\
     & & \mid \epsilon
\end{array}
\]

or in the form of $L=\{(x=x+1)^n;(y=y+1)^n \mid n \geq 0\}$

and vulnerable program $C$:
\begin{lstlisting}
choose n
a=n; b=n
while a>0
    a--
    x = x+1
while b>0
    b--
    y = y+1
\end{lstlisting}

Write the triple for $C$:
\[
% \left\{ \forall l, p, x(p) = x(l) \right\}
\left\{ x(\mathcal{P})=x(L) \wedge y(\mathcal{P}) = y(L)\right\}
\left[
\begin{array}{ll}
p \in \mathcal{P} & , ~C(p) \\
l \in L & , ~l
\end{array}
\right]
\left\{
\begin{array}{c}
\forall l \in L , \exists p \in \mathcal{P} \\
x(p) = x(l) \wedge y(p) = y(l)
\end{array}
\right\}
\]
where $\mathcal{P}:=\mathbb{N}$
